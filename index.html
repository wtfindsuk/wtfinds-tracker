function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function checkKey_(key) { const expected = PropertiesService.getScriptProperties().getProperty('TRACKER_PASSCODE'); if (!expected || typeof key !== 'string' || key !== expected) { Utilities.sleep(1500); return false; } return true; } function doGet(e) { return json_({ ok: false, error: 'unauthorized' }); } function doPost(e) { let payload; try { payload = JSON.parse(e.postData.contents); } catch (err) { return json_({ ok: false, error: 'bad request' }); } if (!checkKey_(payload.key)) return json_({ ok: false, error: 'unauthorized' }); if (payload.action === 'feed') { const url = String(payload.url || ''); const m = url.match(/^https:\/\/([^\/?#]+)[\/?]/); if (!m || FEED_HOSTS.indexOf(m[1].toLowerCase()) < 0) return json_({ ok: false, error: 'feed not allowed' }); const cache = CacheService.getScriptCache(); const cacheKey = 'feed:' + Utilities.base64EncodeWebSafe(url).slice(0, 200); let text = cache.get(cacheKey); if (!text) { const res = UrlFetchApp.fetch(url, { muteHttpExceptions: true, followRedirects: true }); if (res.getResponseCode() !== 200) return json_({ ok: false, error: 'feed returned ' + res.getResponseCode() }); text = res.getContentText(); // Web pages (not .ics calendars): drop scripts, styles, images and the page header so it fits in the cache if (!/BEGIN:VCALENDAR/.test(text)) { text = text .replace(//gi, '') .replace(//gi, '') .replace(//gi, '') .replace(//gi, '') .replace(/]*>/gi, '') .replace(/\s+/g, ' '); } if (text.length < 95000) cache.put(cacheKey, text, 21600); // 6 hours } return json_({ ok: true, text: text }); } const ss = SpreadsheetApp.openById(SHEET_ID); if (payload.action === 'load') { const data = {}; ss.getSheets().forEach(sheet => { data[sheet.getName()] = sheet.getDataRange().getValues(); }); return json_({ ok: true, data }); } if (payload.action === 'save') { const sheetName = payload.sheet, rows = payload.rows; let sheet = ss.getSheetByName(sheetName); if (!sheet) sheet = ss.insertSheet(sheetName); sheet.clearContents(); if (rows && rows.length) sheet.getRange(1, 1, rows.length, rows[0].length).setValues(rows); return json_({ ok: true }); } return json_({ ok: false, error: 'Unknown action' }); }
